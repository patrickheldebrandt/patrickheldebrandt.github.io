`%||%` <- function(x, y) {
  if (is.null(x)) y else x
}

legiscan_get <- function(params) {
  response <- tryCatch(
    GET(
      "https://api.legiscan.com/",
      query = c(key = LEGISCAN_KEY, params)
    ),
    error = function(e) NULL
  )
  
  if (is.null(response) || http_error(response)) {
    return(NULL)
  }
  
  txt <- content(response, as = "text", encoding = "UTF-8")
  
  out <- tryCatch(
    fromJSON(txt, simplifyVector = FALSE),
    error = function(e) NULL
  )
  
  Sys.sleep(0.5)
  out
}

get_sessions_for_state <- function(state_abbr) {
  res <- legiscan_get(list(
    op = "getSessionList",
    state = state_abbr
  ))
  
  if (is.null(res) || is.null(res$sessions)) {
    return(tibble())
  }
  
  map_dfr(res$sessions, function(x) {
    tibble(
      session_id = x$session_id %||% NA_integer_,
      year_start = x$year_start %||% NA_integer_,
      year_end = x$year_end %||% NA_integer_,
      special = x$special %||% NA_integer_,
      session_name = x$session_name %||% NA_character_
    )
  })
}


get_master_list <- function(session_id) {
  res <- legiscan_get(list(
    op = "getMasterList",
    id = session_id
  ))
  
  if (is.null(res) || is.null(res$masterlist)) {
    return(tibble())
  }
  
  items <- res$masterlist
  items <- items[names(items) != "session"]
  
  map_dfr(items, function(x) {
    tibble(
      legiscan_bill_id = x$bill_id %||% NA_integer_,
      number = x$number %||% x$bill_number %||% NA_character_,
      title_api = x$title %||% NA_character_
    )
  }) %>%
    mutate(
      number_clean = str_to_upper(str_replace_all(number, "\\s+", ""))
    )
}


session_cache <- new.env(parent = emptyenv())

get_state_year_master <- function(state_abbr, year) {
  cache_key <- paste(state_abbr, year, sep = "_")
  
  if (exists(cache_key, envir = session_cache)) {
    return(get(cache_key, envir = session_cache))
  }
  
  sessions <- get_sessions_for_state(state_abbr)
  
  if (nrow(sessions) == 0) {
    out <- tibble()
    assign(cache_key, out, envir = session_cache)
    return(out)
  }
  
  possible_sessions <- sessions %>%
    filter(
      year >= year_start,
      year <= year_end
    )
  
  if (nrow(possible_sessions) == 0) {
    possible_sessions <- sessions %>%
      filter(year_start == year | year_end == year)
  }
  
  if (nrow(possible_sessions) == 0) {
    out <- tibble()
    assign(cache_key, out, envir = session_cache)
    return(out)
  }
  
  out <- map_dfr(possible_sessions$session_id, function(sid) {
    get_master_list(sid) %>%
      mutate(session_id = sid)
  })
  
  assign(cache_key, out, envir = session_cache)
  out
}

# Match the bills
find_legiscan_bill_id <- function(state_abbr, year, bill_number) {
  master <- get_state_year_master(state_abbr, year)
  
  if (nrow(master) == 0) {
    return(list(
      id = NA_integer_,
      status = "no session/master list"
    ))
  }
  
  bill_number_clean <- str_to_upper(str_replace_all(bill_number, "\\s+", ""))
  
  matched <- master %>%
    filter(number_clean == bill_number_clean)
  
  if (nrow(matched) == 0) {
    close_matches <- master %>%
      filter(str_detect(number_clean, str_extract(bill_number_clean, "[0-9]+"))) %>%
      slice_head(n = 5) %>%
      pull(number) %>%
      paste(collapse = ", ")
    
    return(list(
      id = NA_integer_,
      status = paste0(
        "no exact match for ", bill_number_clean,
        "; close number matches: ", close_matches
      )
    ))
  }
  
  list(
    id = matched$legiscan_bill_id[1],
    status = paste("matched", matched$number[1], "in session", matched$session_id[1])
  )
}

# find author and partisanship
get_bill_author <- function(legiscan_bill_id) {
  if (is.na(legiscan_bill_id)) {
    return("N/A")
  }
  
  res <- legiscan_get(list(
    op = "getBill",
    id = legiscan_bill_id
  ))
  
  if (is.null(res) || is.null(res$bill) || is.null(res$bill$sponsors)) {
    return("N/A")
  }
  
  sponsors <- res$bill$sponsors
  
  if (length(sponsors) == 0) {
    return("N/A")
  }
  
  sponsor_df <- map_dfr(sponsors, function(x) {
    tibble(
      name = x$name %||% NA_character_,
      sponsor_type_id = x$sponsor_type_id %||% NA_integer_,
      sponsor_order = x$sponsor_order %||% NA_integer_
    )
  }) %>%
    filter(!is.na(name), name != "")
  
  if (nrow(sponsor_df) == 0) {
    return("N/A")
  }
  
  primary <- sponsor_df %>%
    filter(sponsor_type_id == 1 | sponsor_order == 1)
  
  if (nrow(primary) > 0) {
    return(paste(unique(primary$name), collapse = "; "))
  }
  
  paste(unique(sponsor_df$name), collapse = "; ")
}
